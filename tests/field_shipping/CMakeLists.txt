cmake_minimum_required(VERSION 3.25)
project(field_shipping LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(SANITIZE "ASan/UBSan" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(field_current ../../apps/field/field_processor.cpp)
target_include_directories(field_current PUBLIC ../../apps/field)
add_library(field_original ../../research/f01-foundation/reference/field_processor.cpp)
target_compile_definitions(field_original PRIVATE field=field_reference)
add_executable(field_shipping main.cpp)
target_include_directories(field_shipping PRIVATE ../..)
target_link_libraries(field_shipping PRIVATE field_current field_original)
if(APPLE)
 add_library(field_audit SHARED ../realtime_audit_mac.cpp)
 target_link_libraries(field_shipping PRIVATE field_audit)
 target_compile_definitions(field_shipping PRIVATE FIELD_RT_MAC=1)
endif()
enable_testing()
add_test(NAME field_shipping COMMAND field_shipping)
