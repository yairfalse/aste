cmake_minimum_required(VERSION 3.25)
project(impulse_click_sub LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CLICK_SUB_SANITIZERS "Address and undefined behavior checks" OFF)
add_executable(click_sub main.cpp ../../apps/impulse/impulse_processor.cpp)
target_compile_options(click_sub PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CLICK_SUB_SANITIZERS)
  target_compile_options(click_sub PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(click_sub PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME click_sub_sound_contract COMMAND click_sub --check)
