cmake_minimum_required(VERSION 3.25)
project(field_foundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(F01_SANITIZERS "ASan and UBSan" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(F01_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(field_previous reference/field_processor.cpp)
target_include_directories(field_previous PUBLIC reference)
add_executable(f01_lab lab.cpp)
target_include_directories(f01_lab PRIVATE ../..)
target_link_libraries(f01_lab PRIVATE field_previous)
add_executable(f01_contracts tests.cpp)
if(APPLE)
 add_library(f01_audit SHARED ../../tests/realtime_audit_mac.cpp)
 target_link_libraries(f01_contracts PRIVATE f01_audit)
 target_compile_definitions(f01_contracts PRIVATE F01_MAC_AUDIT=1)
endif()
enable_testing()
add_test(NAME f01_contracts COMMAND f01_contracts)
if(APPLE)
 add_executable(f01_au_reference au_reference.cpp)
 target_include_directories(f01_au_reference PRIVATE ../..)
 target_compile_options(f01_au_reference PRIVATE -Wno-deprecated-declarations)
 target_link_libraries(f01_au_reference PRIVATE "-framework AudioUnit" "-framework AudioToolbox" "-framework CoreFoundation")
endif()
add_executable(f01_reference_match reference_match.cpp)
target_include_directories(f01_reference_match PRIVATE ../..)
