cmake_minimum_required(VERSION 3.25)
project(n01_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(N01_SANITIZERS "Audit the offline experiment with ASan/UBSan" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(N01_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(n01_core core.cpp)
target_include_directories(n01_core PUBLIC .)
add_executable(n01_tests tests.cpp)
target_link_libraries(n01_tests PRIVATE n01_core)
enable_testing()
add_test(NAME n01_contracts COMMAND n01_tests)
