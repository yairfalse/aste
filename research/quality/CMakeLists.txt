cmake_minimum_required(VERSION 3.25)
project(aste_quality_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
option(QUALITY_SANITIZERS "Address and undefined behavior checks" OFF)
if(QUALITY_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(BASELINE 8a603dcbc59b52a58637fe6305676efc5a406f80)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
foreach(product density harmonic sequence loop impulse field)
  set(destination "${CMAKE_CURRENT_BINARY_DIR}/reference/${product}")
  file(MAKE_DIRECTORY "${destination}")
  set(files "${product}_processor.cpp" "${product}_processor.hpp")
  if(product STREQUAL loop)
    list(APPEND files tape_snapshots.hpp)
  endif()
  foreach(file IN LISTS files)
    execute_process(COMMAND git show "${BASELINE}:apps/${product}/${file}"
      WORKING_DIRECTORY "${ROOT}" OUTPUT_VARIABLE source
      RESULT_VARIABLE result ERROR_VARIABLE error)
    if(NOT result EQUAL 0)
      message(FATAL_ERROR "Cannot read pinned production reference: ${error}")
    endif()
    string(REPLACE "namespace aste::${product}" "namespace aste::${product}_reference" source "${source}")
    file(WRITE "${destination}/${file}" "${source}")
  endforeach()
  list(APPEND sources "${ROOT}/apps/${product}/${product}_processor.cpp"
                      "${destination}/${product}_processor.cpp")
endforeach()
add_executable(quality_comparison compare.cpp ${sources})
target_include_directories(quality_comparison PRIVATE "${ROOT}" "${CMAKE_CURRENT_BINARY_DIR}")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${ROOT}"
  OUTPUT_VARIABLE COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
target_compile_definitions(quality_comparison PRIVATE QUALITY_COMMIT="${COMMIT}")
enable_testing()
add_test(NAME legacy_audio_and_meter COMMAND quality_comparison --check)
