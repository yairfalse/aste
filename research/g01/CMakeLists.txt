cmake_minimum_required(VERSION 3.25)
project(g01_laboratory LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(G01_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(G01_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(g01_core core.cpp)
target_include_directories(g01_core PUBLIC .)
add_executable(g01_tests tests.cpp)
target_link_libraries(g01_tests PRIVATE g01_core)
if(APPLE)
 add_library(g01_rt_audit SHARED ../../tests/realtime_audit_mac.cpp)
 target_link_libraries(g01_tests PRIVATE g01_rt_audit)
 target_compile_definitions(g01_tests PRIVATE G01_MAC_AUDIT=1)
endif()
add_executable(g01_lab lab.cpp ../../apps/loop/loop_processor.cpp ../../apps/field/field_processor.cpp)
target_include_directories(g01_lab PRIVATE ../..)
target_link_libraries(g01_lab PRIVATE g01_core)
enable_testing()
add_test(NAME g01_contracts COMMAND g01_tests)
add_test(NAME g01_reproducibility COMMAND g01_lab --check)
find_program(G01_NODE_EXECUTABLE node)
if(G01_NODE_EXECUTABLE)
 add_test(NAME g01_ui_model COMMAND "${G01_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/ui_checks.js")
endif()
