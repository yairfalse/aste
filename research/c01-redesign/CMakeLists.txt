cmake_minimum_required(VERSION 3.25)
project(c01_sound_redesign LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(C01_RED_SANITIZERS "ASan/UBSan" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(C01_RED_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(c01_previous reference/core.cpp)
target_include_directories(c01_previous PUBLIC reference)
add_executable(c01_baseline baseline.cpp)
target_include_directories(c01_baseline PRIVATE ../..)
target_link_libraries(c01_baseline PRIVATE c01_previous)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/circuit.cpp")
 add_library(c01_revised circuit.cpp)
 target_include_directories(c01_revised PUBLIC .)
 add_executable(c01_redesign lab.cpp)
 target_include_directories(c01_redesign PRIVATE ../..)
 target_link_libraries(c01_redesign PRIVATE c01_previous c01_revised)
 add_executable(c01_redesign_tests tests.cpp)
 target_link_libraries(c01_redesign_tests PRIVATE c01_revised)
 if(APPLE)
  add_library(c01_red_audit SHARED ../../tests/realtime_audit_mac.cpp)
  target_link_libraries(c01_redesign_tests PRIVATE c01_red_audit)
  target_compile_definitions(c01_redesign_tests PRIVATE C01_RED_MAC_AUDIT=1)
 endif()
 enable_testing()
 add_test(NAME c01_redesign_contracts COMMAND c01_redesign_tests)
endif()
