cmake_minimum_required(VERSION 3.25)
project(c01_research VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(C01_SANITIZERS "Enable address and undefined behavior checks" OFF)
option(C01_PLUGIN "Build the isolated JUCE development VST3" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(C01_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(c01_core core.cpp)
target_include_directories(c01_core PUBLIC .)
add_executable(c01_tests tests.cpp)
target_link_libraries(c01_tests PRIVATE c01_core)
if(APPLE)
  add_library(c01_rt_audit SHARED ../../tests/realtime_audit_mac.cpp)
  target_link_libraries(c01_tests PRIVATE c01_rt_audit)
  target_compile_definitions(c01_tests PRIVATE C01_MAC_AUDIT=1)
endif()
add_executable(c01_lab lab.cpp)
target_link_libraries(c01_lab PRIVATE c01_core)
add_executable(c01_benchmark benchmark.cpp)
target_link_libraries(c01_benchmark PRIVATE c01_core)
add_executable(c01_family family.cpp ../../apps/sequence/sequence_processor.cpp
  ../../apps/field/field_processor.cpp ../../apps/resonator/resonator_processor.cpp)
target_include_directories(c01_family PRIVATE ../../apps/sequence ../../apps/field
  ../../apps/resonator ../../libs/parameters)
target_link_libraries(c01_family PRIVATE c01_core)
enable_testing()
add_test(NAME c01_contracts COMMAND c01_tests)
if(C01_PLUGIN)
  if(NOT C01_JUCE_SOURCE)
    message(FATAL_ERROR "Set C01_JUCE_SOURCE to the repository's pinned JUCE checkout")
  endif()
  add_subdirectory("${C01_JUCE_SOURCE}" "${CMAKE_BINARY_DIR}/juce")
  juce_add_plugin(ContaminationC01
    VERSION 0.1.0 COMPANY_NAME "Aste Internal"
    BUNDLE_ID "invalid.aste.contamination-c01" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE
    COPY_PLUGIN_AFTER_BUILD FALSE PLUGIN_MANUFACTURER_CODE Aste PLUGIN_CODE Cnt1
    FORMATS VST3 VST3_CATEGORIES Instrument Synth PRODUCT_NAME "Contamination C-01")
  target_sources(ContaminationC01 PRIVATE ../../apps/contamination/contamination_plugin.cpp)
  target_include_directories(ContaminationC01 PUBLIC ../../libs/parameters ../../apps/contamination .)
  target_link_libraries(ContaminationC01 PRIVATE c01_core juce::juce_audio_utils)
  target_compile_definitions(ContaminationC01 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_VST3_EMULATE_MIDI_CC_WITH_PARAMETERS=0)
  if(APPLE)
    add_custom_command(TARGET ContaminationC01_VST3 POST_BUILD
      COMMAND codesign --force --deep --sign - "$<TARGET_BUNDLE_DIR:ContaminationC01_VST3>")
  endif()
  add_executable(c01_plugin_tests plugin_tests.cpp)
  target_include_directories(c01_plugin_tests PRIVATE "${C01_JUCE_SOURCE}/modules")
  target_compile_definitions(c01_plugin_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(c01_plugin_tests PRIVATE ContaminationC01)
  if(APPLE)
    target_link_libraries(c01_plugin_tests PRIVATE c01_rt_audit)
    target_compile_definitions(c01_plugin_tests PRIVATE C01_MAC_AUDIT=1)
  endif()
  add_test(NAME c01_plugin_contracts COMMAND c01_plugin_tests)
endif()
