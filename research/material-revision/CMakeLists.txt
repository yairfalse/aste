cmake_minimum_required(VERSION 3.25)
project(aste_material_review LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MATERIAL_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MATERIAL_REFERENCE "${CMAKE_CURRENT_SOURCE_DIR}/reference" CACHE PATH "Frozen 0.7.1 source snapshot")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_reference.py" "${MATERIAL_REFERENCE}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set(PRODUCTS density harmonic sequence loop impulse field non-machine resonator)
add_executable(material_review lab.cpp)
foreach(PRODUCT IN LISTS PRODUCTS)
  string(REPLACE "-" "_" STEM "${PRODUCT}")
  target_sources(material_review PRIVATE "${ROOT}/apps/${PRODUCT}/${STEM}_processor.cpp" "${CMAKE_CURRENT_BINARY_DIR}/reference/${PRODUCT}/${STEM}_processor.cpp")
endforeach()
target_include_directories(material_review PRIVATE "${ROOT}" "${CMAKE_CURRENT_BINARY_DIR}")
if(MATERIAL_SANITIZERS)
  target_compile_options(material_review PRIVATE -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(material_review PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME material_review COMMAND material_review)
