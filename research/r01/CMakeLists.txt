cmake_minimum_required(VERSION 3.25)
project(r01_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
option(R01_SANITIZERS "Audit the isolated research core" OFF)
if(R01_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(r01_core core.cpp)
target_include_directories(r01_core PUBLIC .)
add_executable(r01_tests tests.cpp)
target_link_libraries(r01_tests PRIVATE r01_core)
if(APPLE)
  add_library(r01_rt_audit SHARED ../../tests/realtime_audit_mac.cpp)
  target_link_libraries(r01_tests PRIVATE r01_rt_audit)
  target_compile_definitions(r01_tests PRIVATE R01_MAC_AUDIT=1)
endif()
enable_testing()
add_test(NAME r01_contracts COMMAND r01_tests)
